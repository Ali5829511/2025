### قاعدة البيانات

```sql
-- جدول الملصقات
CREATE TABLE stickers (
    id INTEGER PRIMARY KEY,
    sticker_number TEXT UNIQUE,    -- رقم الملصق الفريد
    resident_id INTEGER,           -- معرف الساكن
    plate_number TEXT,             -- رقم لوحة السيارة
    issue_date DATE,               -- تاريخ الإصدار
    expiry_date DATE,              -- تاريخ الانتهاء
    status TEXT,                   -- الحالة (فعال/منتهي/معلق)
    
    -- الروابط
    FOREIGN KEY (resident_id) REFERENCES residents(id),
    FOREIGN KEY (plate_number) REFERENCES vehicles(plate_number)
);
```
